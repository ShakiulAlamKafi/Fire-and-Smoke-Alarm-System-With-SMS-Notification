#include <LiquidCrystal_I2C.h>
LiquidCrystal_I2C lcd(0x20,6,2);
// C++ code
//
int v_GasSen=0;
int v_TempSens=0;
void setup()
{
  pinMode(A0, INPUT);
  pinMode(7, OUTPUT);
  pinMode(4, OUTPUT);
  pinMode(A1, INPUT);
  pinMode(2, OUTPUT);
  lcd.init();
  lcd.backlight();
  lcd.setCursor(0,0);
  lcd.print("Smoke Detected!");
}

void loop()
{
  //smoke alarm
  v_GasSen=analogRead(A0);
  if(v_GasSen>=250)
  {
    tone(7,523,1000);  //play tone 60(CS= 523 HZ)
    digitalWrite(9, HIGH);
  }
  v_TempSens=-40+0.488155*(analogRead(A1)-20);
  if(v_TempSens>=70)
  {
    tone(7,523,1000);  //play tone 60(CS= 523 HZ)
    digitalWrite(12, HIGH);
  }

  delay(10); // Wait for 10 millisecond(s)
}
